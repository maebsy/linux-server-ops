## 8. 로그 체크포인트 — nginx 로그는 www-data:adm 0640 → cron 실행자가 읽으려면 ACL 부여
$ setfacl -m g:agent-core:r /var/log/nginx/access.log /var/log/nginx/error.log && setfacl -m g:agent-core:rx /var/log/nginx && getfacl -p /var/log/nginx/access.log | grep agent-core

(exit code: 0)

$ sudo -u agent-admin awk '$9 ~ /^[0-9]+$/ { n++; if ($9 >= 500) e++; else if ($9 >= 400) w++ } END { printf "요청 %d, 4xx %d, 5xx %d (5xx 비율 %.1f%%)\n", n, w, e, n ? 100*e/n : 0 }' /var/log/nginx/access.log
요청 6, 4xx 2, 5xx 0 (5xx 비율 0.0%)
(exit code: 0)

$ grep -A3 '^/var/log/nginx' /etc/logrotate.d/nginx | head -n 4; echo '→ nginx 로그는 logrotate(daily, 14개)가, monitor-nginx.log 는 monitor.sh 내장 순환이 관리 — 같은 파일을 두 방식으로 돌리지 않음'
/var/log/nginx/*.log {
	daily
	missingok
	rotate 14
→ nginx 로그는 logrotate(daily, 14개)가, monitor-nginx.log 는 monitor.sh 내장 순환이 관리 — 같은 파일을 두 방식으로 돌리지 않음
(exit code: 0)

$ grep -h 'postrotate' -A3 /etc/logrotate.d/nginx | head -n 4; echo '→ logrotate 가 파일을 바꾸면 ACL 이 사라지므로, 계속 읽어야 하면 /var/log/nginx 에 default ACL(setfacl -d -m g:agent-core:r) 을 함께 둔다'
	postrotate
		invoke-rc.d nginx rotate >/dev/null 2>&1
	endscript
}
→ logrotate 가 파일을 바꾸면 ACL 이 사라지므로, 계속 읽어야 하면 /var/log/nginx 에 default ACL(setfacl -d -m g:agent-core:r) 을 함께 둔다
(exit code: 0)
