# SSH 포트 20022, Root 원격 로그인 차단
# 수집: 2026-10-21 00:54:09 KST · 호스트: agent-server · Ubuntu 22.04.5 LTS

$ cat /etc/ssh/sshd_config.d/00-agent-hardening.conf
# /etc/ssh/sshd_config.d/00-agent-hardening.conf  (setup/01-ssh.sh 가 설치)
#
# 파일 이름을 00- 으로 시작하는 이유: sshd 는 같은 설정이 여러 번 나오면 "처음 읽은 값" 을 쓴다.
# Ubuntu 의 sshd_config 는 맨 위에서 sshd_config.d/*.conf 를 알파벳 순으로 읽으므로,
# 00- 이 50-cloud-init.conf 같은 다른 조각보다 먼저 읽혀 이 값이 확실히 적용된다.
# (단, Port 는 예외로 '누적' 된다. 그래서 01-ssh.sh 가 본문의 다른 Port 줄을 주석 처리한다.)

# 기본 포트(22)를 노리는 무차별 대입 스캔을 대부분 피한다
Port 20022

# root 로 직접 원격 로그인 금지: 일반 계정으로 접속 후 필요할 때만 sudo (누가 무엇을 했는지 기록이 남음)
PermitRootLogin no

$ grep -nE '^[#[:space:]]*(Port|PermitRootLogin)' /etc/ssh/sshd_config
14:#Port 22
33:#PermitRootLogin prohibit-password

## sshd 가 실제로 적용한 값 (sshd -T: 모든 설정 파일을 합친 최종 값)
$ sshd -T 2>/dev/null | grep -Ei '^(port|permitrootlogin) '
port 20022
permitrootlogin no
